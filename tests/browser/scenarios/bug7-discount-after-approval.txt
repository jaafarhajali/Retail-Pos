// Bug 7: the cashier's limit is 10 %. 10 % passes; raising it to 10.5 % must ask for the PIN, even after an earlier approval.
const xText = `[...document.querySelectorAll('#cart .x')].map(e => e.textContent).join(',')`;
const lineDiscount = async (pct) => {
  await click('#btn-qty'); await waitFor(shown('m-line'), 'line dialog'); await sleep(400);
  await type('line-discount', String(pct)); await click('#line-ok'); await sleep(1200);
};
await click('.tile'); await sleep(400);
log('limit on this till:', await ev('window.POS.maxDiscount') + ' % · can discount: ' + await ev('window.POS.can.discount'));
await lineDiscount(10);
log('10 %: PIN asked?', await ev(shown('m-pin')) + ' · cart ' + JSON.stringify(await ev(xText)));
await lineDiscount(10.5);
log('then 10.5 %: PIN asked?', await ev(shown('m-pin')) + ' · cart ' + JSON.stringify(await ev(xText)));
if (await ev(shown('m-pin'))) { await sleep(500); await click('#m-pin .btn-close'); await waitFor(`!${shown('m-pin')}`, 'PIN dialog to close'); await sleep(500); }
log('after cancelling: cart', JSON.stringify(await ev(xText)));

// An approval earlier in the same sale: 15 % approved with the PIN, then raised to 30 % by hand.
await lineDiscount(15);
await waitFor(shown('m-pin'), 'PIN dialog for 15 %'); await sleep(500);
await type('pin-input', '2468'); await click('#pin-ok'); await waitFor(`!${shown('m-pin')}`, 'approval'); await sleep(600);
log('15 % approved: cart', JSON.stringify(await ev(xText)));
await lineDiscount(30);
log('then 30 % after the approval: PIN asked?', await ev(shown('m-pin')) + ' · cart ' + JSON.stringify(await ev(xText)));
if (await ev(shown('m-pin'))) { await sleep(500); await click('#m-pin .btn-close'); await waitFor(`!${shown('m-pin')}`, 'PIN dialog to close'); await sleep(500); }
await lineDiscount(12);
log('then 12 % (under the approved 15 %): PIN asked?', await ev(shown('m-pin')) + ' · cart ' + JSON.stringify(await ev(xText)));
if (await ev(shown('m-pin'))) { await sleep(500); await click('#m-pin .btn-close'); await waitFor(`!${shown('m-pin')}`, 'PIN dialog to close'); await sleep(500); }
